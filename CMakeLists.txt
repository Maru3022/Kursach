cmake_minimum_required(VERSION 3.20)
project(TicTacToe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_GAME "Build the raylib graphical application" ON)
add_library(game_logic src/game.cpp src/ai.cpp src/tournament.cpp src/statistics.cpp)
target_include_directories(game_logic PUBLIC src)
enable_testing()
add_executable(game_tests tests/game_tests.cpp)
target_link_libraries(game_tests PRIVATE game_logic)
add_test(NAME game_logic_tests COMMAND game_tests)
if(BUILD_GAME)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
  add_executable(TicTacToe src/main.cpp src/interface.cpp)
  target_link_libraries(TicTacToe PRIVATE game_logic raylib)
  if(MSVC)
    target_compile_options(TicTacToe PRIVATE /utf-8)
    target_compile_options(game_logic PRIVATE /utf-8)
  endif()
  if(MINGW)
    target_link_options(TicTacToe PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET TicTacToe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:TicTacToe>/assets")
  install(TARGETS TicTacToe RUNTIME DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
  install(FILES README.txt DESTINATION .)
endif()
